from pathlib import Path

# Static skill metadata is NOT trustworthy (SkillCloak bypasses static scanners).
SCANNER_SKIPPED_DIRS = {".cache", "__pycache__", ".git", "node_modules",
                        "test/fixtures", "assets", ".hidden"}
CLOAK_INDICATORS = ("base64.b64decode", "exec(", "eval(", "marshal.loads",
                    "__import__", "getattr(", "compile(")

def audit_skill(skill_dir: str) -> dict:
    p = Path(skill_dir)
    flags = []
    for f in p.rglob("*"):
        if not f.is_file():
            continue
        rel = str(f.relative_to(p))
        # payload hidden in a folder scanners usually skip
        if any(d in rel for d in SCANNER_SKIPPED_DIRS) and f.suffix in (".py", ".js", ".bin"):
            flags.append(f"payload-in-skipped-dir:{rel}")
        try:
            text = f.read_text(errors="ignore")
        except Exception:
            continue
        for ind in CLOAK_INDICATORS:
            if ind in text:
                flags.append(f"runtime-reassembly:{ind}@{rel}")
    trusted = not flags
    return {"skill": p.name, "static_trusted": trusted, "flags": flags,
            "edip_tags": ["NHI-authority", "cloaked-skill"] if flags else [],
            "pipeline": "SSS",
            "decision": "COMPENSATING_CONTROL" if flags else "ALLOW",
            "required_control": None if trusted else
                "Runtime detonation in sandbox before enabling; "
                "never trust static metadata alone"}

if __name__ == "__main__":

01 · Why This Exists — v48 Gap Closure

Business logic flaws — IDORs, broken access control, horizontal privilege escalation, business flow bypasses, and multi-step logic chains — are design-level vulnerabilities with no CVE. Autonomous AI pentest tools like XBOW now surface these at scale by reasoning about expected access patterns before testing them, asking "Should I?" not just "Can I?"

Traditional DAST scanners see a valid 200 OK and mark it passing. They cannot correlate behaviour across multiple authenticated sessions. These findings will never appear in the CISA KEV feed or carry a CVSS score. Without this implementation, Tempris EDIP silently drops them.

This spec closes the gap. BL flaw findings enter via the existing edip_non_cve_extension.py SSS pipeline, get TES-scored, and queue for the same EDIP decision workflow as CVE findings.
